cmake_minimum_required(VERSION 3.16)
project(route_preview LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
if(NOT LVGL_ROOT OR NOT CJSON_ROOT)
  message(FATAL_ERROR "Set LVGL_ROOT and CJSON_ROOT to the firmware managed component sources")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_ROOT}" lvgl)
add_executable(preview preview.cpp "${CJSON_ROOT}/cJSON.c")
target_include_directories(preview PRIVATE "${CJSON_ROOT}" ../route)
target_link_libraries(preview PRIVATE lvgl m)
# Assertions validate navigation, visibility, and overflow even in a Release build.
target_compile_options(preview PRIVATE -UNDEBUG)
